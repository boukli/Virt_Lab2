#!/bin/bash
# Multiple network interfaces

qemu-system-x86_64 \
  -enable-kvm \
  -m 2048 \
  -drive file=vm.qcow2,format=qcow2 \
  \
  # Management network (NAT)
  -netdev user,id=mgmt,hostfwd=tcp::2222-:22 \
  -device virtio-net-pci,netdev=mgmt,mac=52:54:00:12:34:01 \
  \
  # Data network (bridge)
  -netdev bridge,id=data,br=br0 \
  -device virtio-net-pci,netdev=data,mac=52:54:00:12:34:02
