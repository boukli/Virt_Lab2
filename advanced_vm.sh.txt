#!/bin/bash
# Advanced VM configuration with multiple disks and optimized settings
qemu-system-x86_64 \
  -enable-kvm \
  -name "Advanced-VM" \
  -m 4096 \
  -smp 4 \
  -drive file=system.qcow2,if=virtio,format=qcow2,cache=writeback \
  -drive file=data.qcow2,if=virtio,format=qcow2,cache=none \
  -netdev user,id=net0,hostfwd=tcp::2222-:22 \
  -device virtio-net-pci,netdev=net0 \
  -boot order=c
