#!/bin/bash
# Running VM in background with monitor interface
qemu-system-x86_64 \
  -enable-kvm \
  -m 2048 \
  -drive file=vm.qcow2,format=qcow2 \
  -monitor telnet:127.0.0.1:4444,server,nowait \
  -daemonize

# Connect to monitor
telnet 127.0.0.1 4444

# Monitor commands examples:
# info status
# info block
# info network
# savevm snapshot1
# loadvm snapshot1
