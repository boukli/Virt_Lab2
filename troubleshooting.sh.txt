#!/bin/bash
# Performance troubleshooting commands

# Check CPU usage and steal time
mpstat -P ALL 1 5

# Check memory usage
free -h
cat /proc/meminfo

# Check I/O statistics
iostat -x 1 5

# Check network statistics
sar -n DEV 1 5

# Check for CPU overcommit
grep -i commit /proc/meminfo

# Check VM performance counters
virsh domstats vm-name

# Check QEMU process details
ps aux | grep qemu-system
top -p $(pgrep qemu-system)

# Check disk I/O
iotop
