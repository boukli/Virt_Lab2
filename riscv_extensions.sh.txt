#!/bin/bash
# RISC-V with specific extensions

qemu-system-riscv64 \
    -machine virt \
    -cpu rv64,zba=true,zbb=true,zbc=true,zbs=true \
    -smp 2 \
    -m 2G \
    -drive file=vm.qcow2,format=qcow2,if=virtio \
    -netdev user,id=net0 \
    -device virtio-net-device,netdev=net0

# Check available extensions
qemu-system-riscv64 -cpu help | grep rv64
